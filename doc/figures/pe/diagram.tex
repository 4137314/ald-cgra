% figures/pe.tex — processing-element datapath (logical).
\begin{tikzpicture}[node distance=6mm]
  % operand source list
  \node[align=right, font=\scriptsize, color=cWire] (srcs)
        {N, S, E, W\\ const (imm)\\ self ($\reg{r}$)\\ zero};

  % two operand muxes
  \node[mux, right=14mm of srcs, yshift=6mm]  (ma) {mux};
  \node[mux, right=14mm of srcs, yshift=-9mm] (mb) {mux};
  \node[lbl, above=0.5mm of ma] {sel\_a};
  \node[lbl, below=0.5mm of mb] {sel\_b};

  % alu and register
  \node[alu, right=12mm of $(ma)!0.5!(mb)$] (alu) {ALU};
  \node[reg, right=12mm of alu] (reg) {$\reg{r}$};
  \node[blk, below=10mm of alu, minimum width=30mm] (cfg)
        {config word\\{\scriptsize [imm | op | sel\_a | sel\_b]}};

  % wires: sources -> muxes
  \draw[wire] (srcs.east) -- (ma.west);
  \draw[wire] (srcs.east) -- (mb.west);
  % muxes -> alu
  \draw[wire] (ma.east) -- node[lbl,above]{a} ([yshift=3mm]alu.west);
  \draw[wire] (mb.east) -- node[lbl,below]{b} ([yshift=-3mm]alu.west);
  % alu -> reg -> out
  \draw[wire] (alu.east) -- (reg.west);
  \draw[wire] (reg.east) -- node[lbl,above]{dout} ++(12mm,0);

  % step: one clean vertical arrow straight into reg from below; nothing else
  % is routed through this column, so it cannot cross another wire or label.
  \draw[wire] ([yshift=-10mm]reg.south) -- node[lbl,right]{step} (reg.south);
  \draw[wire] ([xshift=8mm,yshift=-10mm]reg.south east)
        -- node[lbl,right]{reset} (reg.south east);

  % self feedback: routed OVER THE TOP on a rail comfortably above ma (and its
  % sel_a label), well clear of the config/op/step wiring below.
  \coordinate (railY) at ($(ma.north)+(0,4mm)$);
  \draw[wire] (reg.north) -- (reg.north |- railY)
        -- (srcs.north |- railY) -- (srcs.north);
  \node[lbl, above] at ($(reg.north|-railY)!0.5!(srcs.north|-railY)$) {self};
  \draw[wire] (alu.north |- railY) -- node[lbl,right]{acc.} (alu.north);

  % config fan-out: both control wires share a trunk placed clear of mb's
  % footprint (to its left), so sel_a's rise to the upper mux never crosses
  % mb's body; sel_b peels off first, sel_a continues higher.
  \coordinate (trunk) at ($(cfg.west)+(-10mm,0)$);
  \draw[wire] (cfg.west) -| (trunk) |- (mb.south);
  \draw[wire] (cfg.west) -| (trunk) |- (ma.south);
  \draw[wire] (cfg.north) -- node[lbl,right]{op} (alu.south);
\end{tikzpicture}
