% figures/pe — processing-element datapath (logical). Configuration fields
% enter as labelled stubs (bit ranges of the 32-bit word), so no control wire
% crosses the datapath. Feedback of r runs on a rail above everything.
\begin{tikzpicture}[x=1mm, y=1mm, font=\scriptsize]
  % operand sources (one bus feeding both 8-way muxes)
  \node[blk, minimum width=17mm, minimum height=22mm, align=center] (src) at (0,0)
    {N, S, E, W\\[0.3ex]imm\\[0.3ex]self $r$\\[0.3ex]zero};

  \node[mux, minimum height=10mm] (ma) at (25, 7) {mux};
  \node[mux, minimum height=10mm] (mb) at (25,-7) {mux};
  \node[alu] (alu) at (47,0) {ALU};
  \node[reg] (reg) at (68,0) {$r$};

  \draw[wire] (src.east |- ma) -- (ma.west);
  \draw[wire] (src.east |- mb) -- (mb.west);
  \draw[wire] (ma.east) -- node[lbl, above] {$a$} (alu.west |- 0,3);
  \draw[wire] (mb.east) -- node[lbl, below] {$b$} (alu.west |- 0,-3);
  \draw[wire] (alu) -- node[lbl, above] {16} (reg);
  \draw[wire] (reg.east) -- node[lbl, above] {dout} ++(11,0);

  % configuration fields as stubs
  \draw[wire] (ma.north) ++(0,5) node[lbl, left] {sel\_a\,[22:20]} -- (ma.north);
  \draw[wire] (mb.south) ++(0,-6) node[lbl, below] {sel\_b\,[25:23]} -- (mb.south);
  \draw[wire] (alu.south) ++(0,-8) node[lbl, below] {op\,[19:16]} -- (alu.south);
  \draw[wire] (src.south) ++(0,-6) node[lbl, below] {imm\,[15:0]} -- (src.south);

  % clock enable and reset of the result register
  \draw[wire] ([xshift=-2mm]reg.south) ++(0,-8) node[lbl, below left=-1mm and -1mm] {step} -- ++(0,8);
  \draw[wire] ([xshift=2.5mm]reg.south) ++(0,-8) node[lbl, below right=-1mm and -1mm] {reset} -- ++(0,8);

  % feedback rail: r to the source bus (self) and to the ALU (MAC/ACC)
  \def\ry{22}
  \draw[cWire, thick] (reg.north) -- (reg.north |- 0,\ry) -- (src.north |- 0,\ry);
  \draw[wire] (src.north |- 0,\ry) -- (src.north);
  \draw[wire] (alu.north |- 0,\ry) -- (alu.north);
  \fill[cWire] (alu.north |- 0,\ry) circle (0.5);
  \node[lbl, above] at (58,\ry) {feedback of $r$};
  \node[lbl, right] at ([yshift=1.5mm]alu.north) {acc.};
\end{tikzpicture}
