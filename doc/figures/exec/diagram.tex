% figures/exec.tex — one element-wise chunk on the wire, before and after the
% fused EXEC transaction. Message-sequence chart: time runs downward, the two
% lifelines are the host and the device, and each arrow carries its byte count.
% The point of the figure is the arrow COUNT (round trips), with the byte
% totals as the secondary reading.
%
% Geometry note: the two panels are kept narrow and deep on purpose. The figure
% is set with \resizebox to one column, so a wide-and-flat layout would be
% scaled DOWN and the annotations would fall below legible size; at roughly
% square it scales up instead.
\begin{tikzpicture}[x=1mm, y=1mm, font=\scriptsize]

  \def\dx{26}                       % host lifeline at 0, device at \dx

  % #1 from-x, #2 to-x, #3 y, #4 label (may be multi-line, \\ separated)
  \newcommand{\msg}[4]{%
    \draw[-{Latex[length=1.6mm]}, cWire, thick] (#1, #3) -- (#2, #3);
    \node[lbl, anchor=south, align=center, inner sep=1.2pt]
         at ($ (#1, #3)!0.5!(#2, #3) $) {#4};
  }
  % #1 label, #2 colour, #3 subtitle
  \newcommand{\panelhead}[3]{%
    \node[anchor=west, font=\scriptsize\bfseries, #2] at (-6, 21) {#1};
    \node[anchor=west, lbl]                           at (-6, 16) {#3};
    \node[lbl, anchor=south] at (0,    8) {host};
    \node[lbl, anchor=south] at (\dx,  8) {device};
    \draw[cWire!35] (0,   7) -- (0,   -44);
    \draw[cWire!35] (\dx, 7) -- (\dx, -44);
  }

  % ================= protocol v2: three round trips =====================
  \begin{scope}
    \panelhead{protocol v2}{cWire}{3 round trips, 56\,B}

    \msg{0}{\dx}{2}    {\cmd{WR} $+$ inputs $+$ ck\\[-0.3ex] 18\,B}
    \msg{\dx}{0}{-6}   {\cmd{ACK}\;\;1\,B}
    \msg{0}{\dx}{-14}  {\cmd{RUN} $N$\;\;2\,B}
    \msg{\dx}{0}{-22}  {\cmd{ACK}\;\;1\,B}
    \msg{0}{\dx}{-30}  {\cmd{RD}\;\;1\,B}
    \msg{\dx}{0}{-38}  {16 regs $+$ ck\;\;33\,B}

    \foreach \a/\b in {2.5/-6.5, -13.5/-22.5, -29.5/-38.5} {
      \draw[decorate, decoration={brace, amplitude=2.5pt}, cWire!55]
        (-3, \a) -- (-3, \b);
    }
    \node[lbl, rotate=90, anchor=south] at (-6, -18) {three turnarounds};
  \end{scope}

  % ================= protocol v3: one fused round trip ==================
  \begin{scope}[xshift=50mm]
    \panelhead{protocol v3}{cRegE}{1 round trip, 32\,B}

    \msg{0}{\dx}{2}   {\cmd{EXEC} $+$ steps, flags,\\[-0.3ex]
                       mask, inputs $+$ ck\;\;22\,B}
    \msg{\dx}{0}{-6}  {4 taps $+$ ck $+$ status\\[-0.3ex] 10\,B}

    \draw[decorate, decoration={brace, amplitude=2.5pt}, cRegE] (-3, 2.5) -- (-3, -6.5);
    \node[lbl, rotate=90, anchor=south, cRegE] at (-6, -2) {one turnaround};

    % the two exchanges the fusion removed, shown as the space they leave
    \draw[cRegE!45, dashed] (-1, -12) -- (\dx + 1, -12);
    \node[lbl, cRegE, anchor=north, align=center, text width=27mm] at (13, -15)
      {\itshape the four exchanges below\\[-0.3ex]
       this line no longer happen};
  \end{scope}

\end{tikzpicture}
