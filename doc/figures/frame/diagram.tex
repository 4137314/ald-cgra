  % #1 x0, #2 width, #3 fill, #4 edge, #5 label, #6 bytes, #7 offset
  \newcommand{\fld}[7]{%

    \node[align=center, inner sep=0pt] at (#1 + #2/2, 4.9) {#5};
    \node[lbl, inner sep=0pt] at (#1 + #2/2, 1.9) {#6};
    \node[lbl, anchor=south, inner sep=1pt] at (#1 + #2/2, 8) {#7};
  }
  \fld{0}{7}{cWire!10}{cWire!60}{\texttt{7E}}{flag}{}
  \fld{7}{7}{cBlock}{cBlockE}{rev}{1\,B}{0}
  \fld{14}{7}{cBlock}{cBlockE}{op}{1\,B}{1}

  \fld{32}{9}{cState}{cStateE}{seq}{2\,B}{6--7}
  \fld{41}{9}{cBlock}{cBlockE}{len}{2\,B}{8--9}
